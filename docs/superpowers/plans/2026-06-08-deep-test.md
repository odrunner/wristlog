# Deep Test — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A flag-gated (`deep_test`) "Deep Test" button that runs 3–6 mic measurements, harvests the cleanest stable sub-run chunks from each, pools them, and reports a median rate ± STD with a confidence chip — saved to history and to a `deep_test_chunks` table for tuning.

**Architecture:** A pure `extractCleanChunks(samples, opts)` (in `wrotate_test.js`, unit-tested, inline-mirrored VERBATIM) segments one run's tick stream into calm chunks (skips warm-up + disturbances) using the existing `_q2Ls` slope helper. A `medianStd(arr)` pure helper aggregates. `runDeepTest()` orchestrates the run loop (mirrors the batch loop), pools chunks across runs, computes median±STD live, persists, and renders progress/result. New flag `deep_test`; thresholds via `dt_*` localStorage knobs (read with the existing `_q2Num`).

**Tech Stack:** Vanilla JS, Vitest, Playwright (mocked), Supabase. Pure logic mirrored inline (mirror-drift VERBATIM).

// Median + population std of a numeric array (rounded). { median, std, n }.
export function medianStd(arr) {
  const a = (arr || []).filter(x => isFinite(x));
  const n = a.length;
  if (!n) return { median: null, std: null, n: 0 };
  const s = [...a].sort((x, y) => x - y);
  const m = s.length % 2 ? s[(s.length - 1) / 2] : (s[s.length / 2 - 1] + s[s.length / 2]) / 2;
  const mean = a.reduce((x, y) => x + y, 0) / n;
  const std = Math.sqrt(a.reduce((x, y) => x + (y - mean) ** 2, 0) / n);
  return { median: Math.round(m * 10) / 10, std: Math.round(std * 100) / 100, n };
}

// Harvest calm sub-run segments ("chunks") from a tick stream [{t,cd}], skipping the warm-up
// transient and disturbances. Returns [{t0,t1,rate,residualSd,nTicks}] (rate = robust slope ×86.4).
export function extractCleanChunks(samples, opts = {}) {
  const o = { warmupLook: 20, warmupEps: 0.5, warmupFloor: 15, win: 10,
              bandSday: 3, segMinSec: 15, residualMaxMs: 1.5, ...opts };
  const pts = (samples || []).filter(p => p && isFinite(p.t) && isFinite(p.cd));
  if (pts.length < 8) return [];
  const start = pts[0].t, lastT = pts[pts.length - 1].t;
  // 1) warm-up end: first time the from-start rate settles, floored at warmupFloor.
  let settleT = null;
  for (let t = start + o.warmupLook + 1; t <= lastT + 1e-6; t += 1) {
    const a = _q2Ls(pts, start, t), b = _q2Ls(pts, start, t - o.warmupLook);
    if (a != null && b != null && Math.abs(a - b) <= o.warmupEps) { settleT = t; break; }
  }
  const warmupEnd = settleT != null ? Math.max(settleT, start + o.warmupFloor) : start + o.warmupFloor;
  // 2) windowed local rate from warm-up end.
  const series = [];
  for (let t = warmupEnd + o.win; t <= lastT + 1e-6; t += 1) {
    const r = _q2Ls(pts, t - o.win, t);
    if (r != null) series.push({ t, r });
  }
  // 3) segment into calm stretches; break (drop a point) on a disturbance.
  const med = a => { const s = [...a].sort((x, y) => x - y); const m = s.length >> 1; return s.length % 2 ? s[m] : (s[m - 1] + s[m]) / 2; };
  const chunks = [];
  let seg = [];
  const flush = () => {
    if (seg.length >= 2) {
      const t0 = seg[0].t - o.win, t1 = seg[seg.length - 1].t;
      if (t1 - t0 >= o.segMinSec) {
        const rate = _q2Ls(pts, t0, t1);
        if (rate != null) {
          const w = pts.filter(p => p.t >= t0 && p.t <= t1);
          const slopeMs = rate / 86.4;
          const mt = w.reduce((s, p) => s + p.t, 0) / w.length;
          const mc = w.reduce((s, p) => s + p.cd, 0) / w.length;
          const intercept = mc - slopeMs * mt;
          const resStd = Math.sqrt(w.reduce((s, p) => { const e = p.cd - (slopeMs * p.t + intercept); return s + e * e; }, 0) / w.length);
          if (resStd <= o.residualMaxMs) {
            chunks.push({ t0: Math.round(t0 * 10) / 10, t1: Math.round(t1 * 10) / 10, rate: Math.round(rate * 10) / 10, residualSd: Math.round(resStd * 1000) / 1000, nTicks: w.length });
          }
        }
      }
    }
    seg = [];
  };
  for (const pt of series) {
    if (!seg.length) { seg.push(pt); continue; }
    if (Math.abs(pt.r - med(seg.map(s => s.r))) <= o.bandSday) seg.push(pt);
    else flush();
  }
  flush();
  return chunks;
}
```

let _deepTestActive = false;
const _DEEP_TEST_LABEL = 'Deep Test (3–6 runs · median ± STD)';
function _deepRender(text, color) {
  const host = document.getElementById('msr-scatter-plot')?.parentElement;
  if (!host) return;
  let el = document.getElementById('deep-test-readout');
  if (!el) { el = document.createElement('div'); el.id = 'deep-test-readout'; el.style.cssText = 'font-family:monospace;font-size:.72rem;padding:.3rem .4rem;line-height:1.4;'; host.insertAdjacentElement('afterend', el); }
  el.textContent = text; el.style.color = color || '#6ee7b7';
}
async function runDeepTest() {
  const btn = document.getElementById('deep-test-btn');
  if (_deepTestActive) { _deepTestActive = false; if (btn) btn.textContent = 'Cancelling…'; return; }
  if (!_tgHasNative()) { toast('Deep Test needs the native app'); return; }
  const sel = document.getElementById('msr-bph-select');
  const bph = sel ? (parseInt(sel.value) || 0) : 0;
  if (bph === 0) { toast('Pick the watch’s BPH first'); return; }
  if (localStorage.getItem('tg_input_source') === 'piezo') localStorage.setItem('tg_input_source', 'mic');
  const watchId = document.getElementById('msr-watch-select')?.value || null;
  const minRuns = _q2Num('dt_min_runs', 3), maxRuns = _q2Num('dt_max_runs', 6);
  const runMs = _q2Num('dt_run_secs', 90) * 1000, GAP = 4000, RESET = 1200;
  const opts = {};
  for (const k of ['warmupLook', 'warmupEps', 'warmupFloor', 'win', 'bandSday', 'segMinSec', 'residualMaxMs']) {
    const v = parseFloat(localStorage.getItem('dt_' + k)); if (isFinite(v)) opts[k] = v;
  }
  const deepTestId = crypto.randomUUID();
  const sleep = ms => new Promise(r => setTimeout(r, ms));
  const pooledRates = []; const runBeats = [];
  _deepTestActive = true;
  if (btn) btn.textContent = 'Deep Test starting…';
  try {
    for (let run = 0; run < maxRuns && _deepTestActive; run++) {
      if (_msrListening) { stopMsrListen('deep_reset'); await sleep(RESET); }
      if (!_deepTestActive) break;
      _deepRender(`Run ${run + 1}/${maxRuns} measuring… (${pooledRates.length} clean chunks so far)`, '#9aa');
      toggleMsrListen();
      await sleep(runMs);
      if (_msrListening) stopMsrListen('deep');
      const chunks = extractCleanChunks(_msrScatterData, opts);
      chunks.forEach((c, ci) => {
        if (currentUser) db.from('deep_test_chunks').insert({
          deep_test_id: deepTestId, user_id: currentUser.id, watch_id: watchId, bph,
          run_idx: run, chunk_idx: ci, t0: c.t0, t1: c.t1, rate: c.rate, residual_sd: c.residualSd, n_ticks: c.nTicks,
        }).then(({ error }) => { if (error) console.warn('[DEEP] chunk insert failed', error.message); });
        pooledRates.push(c.rate);
      });
      if (typeof _msrLastBe !== 'undefined' && _msrLastBe != null) runBeats.push(_msrLastBe);
      const agg = medianStd(pooledRates);
      const so = agg.median != null ? `${agg.median >= 0 ? '+' : ''}${agg.median} ±${agg.std}` : '—';
      _deepRender(`Run ${run + 1}/${maxRuns} · ${pooledRates.length} chunks · ${so} s/day`, '#9aa');
      // early-out (default off): stop once tight enough
      const earlyStd = _q2Num('dt_early_std', 0);
      if (earlyStd > 0 && run + 1 >= minRuns && agg.n >= 6 && agg.std != null && agg.std <= earlyStd) break;
      await sleep(GAP);
    }
  } finally {
    _deepTestActive = false;
    if (btn) btn.textContent = _DEEP_TEST_LABEL;
  }
  const agg = medianStd(pooledRates);
  const beMed = medianStd(runBeats).median;
  if (agg.median == null) { _deepRender('Deep Test: no clean chunks — try repositioning', '#fbbf24'); toast('Deep Test: no clean data'); return; }
  const good = agg.std != null && agg.std <= 2 && agg.n >= 4;
  _deepRender(`Deep Test: ${agg.median >= 0 ? '+' : ''}${agg.median} ±${agg.std} s/day · ${agg.n} chunks` + (beMed != null ? ` · BE ${beMed}ms` : ''), good ? '#34d399' : '#fbbf24');
  if (currentUser) {
    db.from('timegrapher_results').insert({
      user_id: currentUser.id, watch_id: watchId, rate: agg.median, rate_std: agg.std,
      beat_error: beMed, bph, source: 'deep_test', notes: `Deep Test: ${agg.n} chunks / pooled`,
    }).then(({ error }) => { if (error) console.warn('[DEEP] result insert failed', error.message); else toast('Deep Test saved'); });
  }
}
```
